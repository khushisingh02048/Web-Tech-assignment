// JavaScript Array Methods Demonstration // push(), pop(), shift(), unshift() // Create an array let fruits = ["Apple", "Banana", "Mango"]; console.log("Original Array:", fruits); // ---------------- push() ---------------- // Adds element(s) at the END of the array fruits.push("Orange"); console.log("After push('Orange'):", fruits); // ---------------- pop() ---------------- // Removes the LAST element from the array fruits.pop(); console.log("After pop():", fruits); // ---------------- unshift() ---------------- // Adds element(s) at the START of the array fruits.unshift("Grapes"); console.log("After unshift('Grapes'):", fruits); // ---------------- shift() ---------------- // Removes the FIRST element from the array fruits.shift(); console.log("After shift():", fruits); /* Summary (in code comments): push() -> add element at end pop() -> remove element from end unshift() -> add element at start shift() -> remove element from start */
